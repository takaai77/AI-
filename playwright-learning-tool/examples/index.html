🔍 Google検索を自動化

初級 検索

Googleで検索して結果のスクリーンショットを撮る基本的な例

const { chromium } = require('playwright'); (async () => { // ブラウザを起動（画面を表示） const browser = await chromium.launch({ headless: false }); const page = await browser.newPage(); // Googleにアクセス await page.goto('https://www.google.com'); // 検索ボックスに入力 await page.locator('input[name="q"]').fill('Playwright 入門'); // Enterキーで検索実行 await page.locator('input[name="q"]').press('Enter'); // 結果が表示されるまで待つ await page.waitForLoadState('networkidle'); // スクリーンショットを撮影 await page.screenshot({ path: 'search-result.png' }); console.log('✅ 検索完了！search-result.png を確認してね'); await browser.close(); })();

🌐 複数サイトを巡回してスクリーンショット

初級 繰り返し

複数のWebサイトを順番に訪問してスクリーンショットを撮る

const { chromium } = require('playwright'); (async () => { const browser = await chromium.launch(); const page = await browser.newPage(); // 訪問するサイトのリスト const sites = [ { url: 'https://www.google.com', name: 'google' }, { url: 'https://www.yahoo.co.jp', name: 'yahoo' }, { url: 'https://www.youtube.com', name: 'youtube' } ]; for (const site of sites) { console.log(`📸 ${site.name} を撮影中...`); await page.goto(site.url); await page.screenshot({ path: `${site.name}.png`, fullPage: true }); } console.log('🎉 全サイトの撮影完了！'); await browser.close(); })();

📰 ニュース見出しを収集

中級 スクレイピング

ニュースサイトから見出しを自動収集してファイルに保存

const { chromium } = require('playwright'); const fs = require('fs'); (async () => { const browser = await chromium.launch(); const page = await browser.newPage(); // Yahoo!ニュースにアクセス await page.goto('https://news.yahoo.co.jp'); // 見出しを全て取得 const headlines = await page.locator('article a').allTextContents(); // 空でないものだけフィルタリング const validHeadlines = headlines .filter(h => h.trim().length > 5) .slice(0, 10); // コンソールに表示 console.log('📰 今日のニュース TOP 10:'); validHeadlines.forEach((h, i) => { console.log(`${i + 1}. ${h}`); }); // ファイルに保存 const content = validHeadlines.join('\n'); fs.writeFileSync('headlines.txt', content); console.log('💾 headlines.txt に保存しました'); await browser.close(); })();

📝 フォーム自動入力テンプレート

中級 フォーム

会員登録フォームなどを自動入力するテンプレート

const { chromium } = require('playwright'); (async () => { const browser = await chromium.launch({ headless: false }); const page = await browser.newPage(); // 入力データ（ここを変更） const formData = { name: '山田 太郎', email: 'taro@example.com', password: 'SecurePass123!', phone: '090-1234-5678' }; // フォームページにアクセス（URLを変更） await page.goto('https://example.com/register'); // テキスト入力 await page.getByLabel('お名前').fill(formData.name); await page.getByLabel('メールアドレス').fill(formData.email); await page.getByLabel('パスワード').fill(formData.password); await page.getByLabel('電話番号').fill(formData.phone); // チェックボックス await page.getByLabel('利用規約に同意する').check(); // スクリーンショットで確認 await page.screenshot({ path: 'form-filled.png' }); console.log('✅ フォーム入力完了！'); // 送信はコメントアウト（必要なら有効化） // await page.getByRole('button', { name: '送信' }).click(); await page.waitForTimeout(5000); await browser.close(); })();

📄 ページをPDFとして保存

中級 PDF

WebページをPDFファイルとして保存する

const { chromium } = require('playwright'); (async () => { const browser = await chromium.launch(); const page = await browser.newPage(); // PDFにしたいページにアクセス await page.goto('https://example.com'); // PDFとして保存 await page.pdf({ path: 'page.pdf', format: 'A4', printBackground: true }); console.log('📄 PDFを保存しました！'); await browser.close(); })();
