🏗️ Page Object設計

カートのテストでは、保守性を高めるためにPage Object Modelを使用します。

CartPage クラス

// pages/CartPage.js class CartPage { constructor(page) { this.page = page; // セレクタ定義 this.cartItems = '[data-testid="cart-item"]'; this.quantityInput = '[data-testid="quantity-input"]'; this.removeButton = '[data-testid="remove-item"]'; this.subtotal = '[data-testid="subtotal"]'; this.tax = '[data-testid="tax"]'; this.total = '[data-testid="total"]'; this.checkoutButton = '[data-testid="checkout-btn"]'; this.emptyCartMessage = '[data-testid="empty-cart"]'; } // カートページに移動 async goto() { await this.page.goto('/cart'); await this.page.waitForLoadState('networkidle'); } // カート内の商品数を取得 async getItemCount() { return await this.page.locator(this.cartItems).count(); } // 商品の数量を変更 async changeQuantity(itemIndex, quantity) { const item = this.page.locator(this.cartItems).nth(itemIndex); await item.locator(this.quantityInput).fill(String(quantity)); await item.locator(this.quantityInput).press('Enter'); // 価格更新を待つ await this.page.waitForResponse(resp => resp.url().includes('/api/cart') && resp.status() === 200 ); } // 商品を削除 async removeItem(itemIndex) { const item = this.page.locator(this.cartItems).nth(itemIndex); await item.locator(this.removeButton).click(); await this.page.waitForResponse(resp => resp.url().includes('/api/cart') ); } // 合計金額を取得 async getTotal() { const text = await this.page.locator(this.total).textContent(); return parseInt(text.replace(/[^0-9]/g, '')); } // 税額を取得 async getTax() { const text = await this.page.locator(this.tax).textContent(); return parseInt(text.replace(/[^0-9]/g, '')); } // 購入手続きへ async proceedToCheckout() { await this.page.locator(this.checkoutButton).click(); await this.page.waitForURL('**/checkout**'); } // カートが空かチェック async isEmpty() { return await this.page.locator(this.emptyCartMessage).isVisible(); } } module.exports = { CartPage };

🧪 テストコード実装

1. 基本的なカート操作テスト

// tests/cart.spec.js const { test, expect } = require('@playwright/test'); const { CartPage } = require('../pages/CartPage'); const { ProductPage } = require('../pages/ProductPage'); test.describe('カート操作テスト', () => { let cartPage; let productPage; test.beforeEach(async ({ page }) => { cartPage = new CartPage(page); productPage = new ProductPage(page); }); test('商品をカートに追加できる', async ({ page }) => { // 商品ページにアクセス await productPage.goto('SKU-12345'); // カートに追加 await productPage.addToCart(); // カートページで確認 await cartPage.goto(); const itemCount = await cartPage.getItemCount(); expect(itemCount).toBe(1); }); test('数量を変更すると金額が正しく再計算される', async ({ page }) => { // 事前に商品をカートに追加 await productPage.goto('SKU-12345'); // 単価1,000円の商品 await productPage.addToCart(); await cartPage.goto(); // 数量を3に変更 await cartPage.changeQuantity(0, 3); // 合計金額を検証（税込10%） const total = await cartPage.getTotal(); expect(total).toBe(3300); // 1000 × 3 × 1.1 }); test('商品を削除するとカートから消える', async ({ page }) => { // 2つの商品を追加 await productPage.goto('SKU-12345'); await productPage.addToCart(); await productPage.goto('SKU-67890'); await productPage.addToCart(); await cartPage.goto(); expect(await cartPage.getItemCount()).toBe(2); // 1つ削除 await cartPage.removeItem(0); expect(await cartPage.getItemCount()).toBe(1); }); test('全商品削除でカートが空になる', async ({ page }) => { await productPage.goto('SKU-12345'); await productPage.addToCart(); await cartPage.goto(); await cartPage.removeItem(0); expect(await cartPage.isEmpty()).toBe(true); }); });

2. 在庫・購入制限テスト

test.describe('在庫・購入制限テスト', () => { test('在庫数以上は追加できない', async ({ page }) => { // 在庫5個の商品 await productPage.goto('SKU-LIMITED'); await productPage.addToCart(); await cartPage.goto(); // 在庫を超える数量に変更 await cartPage.changeQuantity(0, 10); // エラーメッセージを確認 const errorMsg = page.locator('[data-testid="stock-error"]'); await expect(errorMsg).toContainText('在庫が不足'); // 数量が在庫上限に戻ることを確認 const quantity = await cartPage.getQuantity(0); expect(quantity).toBeLessThanOrEqual(5); }); test('購入個数制限が機能する', async ({ page }) => { // 1人2個までの商品 await productPage.goto('SKU-LIMIT2'); await productPage.addToCart(); await cartPage.goto(); await cartPage.changeQuantity(0, 5); // 制限メッセージを確認 const limitMsg = page.locator('[data-testid="limit-warning"]'); await expect(limitMsg).toContainText('お一人様2点まで'); }); test('カート保持中に在庫切れになった場合の挙動', async ({ page, context }) => { // 商品をカートに追加 await productPage.goto('SKU-LASTONELEFT'); await productPage.addToCart(); // 別のブラウザで同じ商品を購入（在庫切れを模擬） const page2 = await context.newPage(); await page2.goto('/api/test/deplete-stock/SKU-LASTONELEFT'); // 元のページでカートをリロード await cartPage.goto(); // 在庫切れ通知を確認 const outOfStockBadge = page.locator('[data-testid="out-of-stock-badge"]'); await expect(outOfStockBadge).toBeVisible(); // 購入ボタンが無効になっていることを確認 const checkoutBtn = page.locator('[data-testid="checkout-btn"]'); await expect(checkoutBtn).toBeDisabled(); }); });

test.describe('価格計算テスト', () => { test('消費税が正しく計算される（10%）', async ({ page }) => { // 税抜1,000円の商品 await productPage.goto('SKU-TAX10'); await productPage.addToCart(); await cartPage.goto(); const subtotal = await cartPage.getSubtotal(); const tax = await cartPage.getTax(); const total = await cartPage.getTotal(); expect(subtotal).toBe(1000); expect(tax).toBe(100); expect(total).toBe(1100); }); test('軽減税率が正しく計算される（8%）', async ({ page }) => { // 食品（軽減税率8%） await productPage.goto('SKU-FOOD'); // 税抜1,000円 await productPage.addToCart(); await cartPage.goto(); const tax = await cartPage.getTax(); expect(tax).toBe(80); }); test('クーポン割引が正しく適用される', async ({ page }) => { await productPage.goto('SKU-12345'); // 10,000円 await productPage.addToCart(); await cartPage.goto(); // 10%OFFクーポンを適用 await cartPage.applyCoupon('SAVE10'); const discount = await cartPage.getDiscount(); expect(discount).toBe(1000); const total = await cartPage.getTotal(); expect(total).toBe(9900); // (10000 - 1000) × 1.1 }); test('ポイント利用が正しく計算される', async ({ page }) => { // ログインしてポイントを持っているユーザー await loginAsUser('user-with-points@test.com'); await productPage.goto('SKU-12345'); // 10,000円 await productPage.addToCart(); await cartPage.goto(); // 500ポイント使用 await cartPage.usePoints(500); const total = await cartPage.getTotal(); expect(total).toBe(10500); // 11000 - 500 }); });

⚠️ 実務でよくあるバグと対策

バグ1: 並行操作での数量不整合

複数タブで同時にカート操作すると、数量が正しく反映されないことがある。

対策: APIレスポンスの待機、楽観的ロックのテストを追加

バグ2: 小数点以下の金額計算

消費税計算で小数点が発生した際の丸め処理が不正確。

対策: 端数処理（切り捨て/切り上げ/四捨五入）のテストケースを網羅

バグ3: セッション切れ後のカート消失

ログイン前に追加した商品が、ログイン後に消えてしまう。

対策: ログイン前後のカートマージテストを追加

🏆 本番運用でのベストプラクティス

1. 定期的な合成監視

本番環境で5分ごとにカート追加→削除のテストを自動実行し、障害を早期検知。

2. セール前の負荷テスト

大型セール前に、想定アクセス数の3倍でカート操作の負荷テストを実施。

3. A/Bテストとの連携

カートUIのA/Bテスト時は、両パターンで同じテストが通ることを確認。
