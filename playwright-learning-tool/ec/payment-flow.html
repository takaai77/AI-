💳 EC決済フローのE2Eテスト

100億円規模ECの決済を守る堅牢なテスト設計

🎯 このレッスンで学ぶこと

  • 決済フロー全体のE2Eテスト設計
  • クレジットカード・各種決済のテスト手法
  • 外部決済APIのモック戦略
  • 決済エラーハンドリングのテスト
  • 本番環境に影響しないテスト設計

🔄 EC決済フロー全体像

カート → 配送先入力 → 支払い方法選択 → 確認画面 → 決済処理 → 完了画面

⚠️ 100億ECでの決済テスト鉄則

  • 本番の決済APIには絶対に接続しない
  • テスト用のサンドボックス環境を使用
  • テストデータは必ずクリーンアップ
  • 決済金額・在庫の整合性を必ず検証

🏗️ 決済テスト用Page Object

// pages/CheckoutPage.js class CheckoutPage { constructor(page) { this.page = page; // 配送先フォーム this.shippingForm = { name: '[data-testid="shipping-name"]', postal: '[data-testid="shipping-postal"]', address: '[data-testid="shipping-address"]', phone: '[data-testid="shipping-phone"]' }; // 支払い方法 this.paymentMethods = { credit: '[data-testid="payment-credit"]', convenience: '[data-testid="payment-convenience"]', cod: '[data-testid="payment-cod"]', amazon: '[data-testid="payment-amazon-pay"]', paypay: '[data-testid="payment-paypay"]' }; // クレジットカードフォーム this.creditForm = { number: '[data-testid="card-number"]', expiry: '[data-testid="card-expiry"]', cvv: '[data-testid="card-cvv"]', name: '[data-testid="card-name"]' }; // ボタン・確認 this.submitBtn = '[data-testid="submit-order"]'; this.confirmBtn = '[data-testid="confirm-order"]'; this.orderTotal = '[data-testid="order-total"]'; this.orderNumber = '[data-testid="order-number"]'; } // 配送先情報を入力 async fillShippingInfo(info) { await this.page.locator(this.shippingForm.name).fill(info.name); await this.page.locator(this.shippingForm.postal).fill(info.postal); await this.page.locator(this.shippingForm.address).fill(info.address); await this.page.locator(this.shippingForm.phone).fill(info.phone); } // 支払い方法を選択 async selectPaymentMethod(method) { await this.page.locator(this.paymentMethods[method]).click(); } // クレジットカード情報を入力 async fillCreditCard(card) { await this.page.locator(this.creditForm.number).fill(card.number); await this.page.locator(this.creditForm.expiry).fill(card.expiry); await this.page.locator(this.creditForm.cvv).fill(card.cvv); await this.page.locator(this.creditForm.name).fill(card.name); } // 注文を確定 async submitOrder() { await this.page.locator(this.submitBtn).click(); await this.page.locator(this.confirmBtn).click(); } // 注文完了を確認 async getOrderNumber() { await this.page.locator(this.orderNumber).waitFor(); return await this.page.locator(this.orderNumber).textContent(); } // 合計金額を取得 async getOrderTotal() { const text = await this.page.locator(this.orderTotal).textContent(); return parseInt(text.replace(/[^0-9]/g, '')); } }

シナリオ1: クレジットカード決済の正常系

import { test, expect } from '@playwright/test'; import { CheckoutPage } from './pages/CheckoutPage'; import { CartPage } from './pages/CartPage'; // テスト用クレジットカード（決済代行のサンドボックス用） const TEST_CARDS = { visa: { number: '4111111111111111', expiry: '12/28', cvv: '123', name: 'TEST USER' }, mastercard: { number: '5500000000000004', expiry: '12/28', cvv: '123', name: 'TEST USER' }, // 決済失敗用テストカード declined: { number: '4000000000000002', expiry: '12/28', cvv: '123', name: 'TEST USER' } }; test.describe('クレジットカード決済', () => { test('VISAカードで正常に決済できる', async ({ page }) => { const cart = new CartPage(page); const checkout = new CheckoutPage(page); // 商品をカートに追加 await page.goto('/products/test-item'); await page.getByRole('button', { name: 'カートに入れる' }).click(); await page.goto('/cart'); // 購入手続きへ await page.getByRole('button', { name: '購入手続きへ' }).click(); // 配送先入力 await checkout.fillShippingInfo({ name: 'テスト 太郎', postal: '100-0001', address: '東京都千代田区1-1-1', phone: '090-0000-0000' }); // クレジットカード決済を選択 await checkout.selectPaymentMethod('credit'); await checkout.fillCreditCard(TEST_CARDS.visa); // 注文前の合計金額を記録 const expectedTotal = await checkout.getOrderTotal(); // 注文確定 await checkout.submitOrder(); // 注文完了を確認 const orderNumber = await checkout.getOrderNumber(); expect(orderNumber).toMatch(/^ORD-\d{10}$/); // 完了画面のURLを確認 await expect(page).toHaveURL(/.*\/order\/complete/); // 注文履歴に反映されていることを確認 await page.goto('/mypage/orders'); await expect(page.getByText(orderNumber)).toBeVisible(); }); });

test('カード決済が拒否された場合、エラーメッセージが表示される', async ({ page }) => { const checkout = new CheckoutPage(page); // カートに商品追加〜配送先入力（省略） // ... // 決済失敗用カードを使用 await checkout.selectPaymentMethod('credit'); await checkout.fillCreditCard(TEST_CARDS.declined); // 注文を試行 await page.locator('[data-testid="submit-order"]').click(); await page.locator('[data-testid="confirm-order"]').click(); // エラーメッセージを確認 await expect(page.getByText('カード決済に失敗しました')).toBeVisible(); await expect(page.getByText('別のお支払い方法をお選びください')).toBeVisible(); // 注文は完了していないこと await expect(page).not.toHaveURL(/.*\/order\/complete/); // カートの商品は残っていること await page.goto('/cart'); await expect(page.locator('[data-testid="cart-item"]')).toHaveCount(1); });

シナリオ3: 外部決済APIのモック

// 外部決済APIをモックして高速・安定テスト test('PayPay決済（APIモック）', async ({ page }) => { // PayPay APIレスポンスをモック await page.route('**/api/paypay/**', async (route) => { const url = route.request().url(); if (url.includes('/create-payment')) { // 決済作成成功レスポンス await route.fulfill({ status: 200, contentType: 'application/json', body: JSON.stringify({ resultInfo: { code: 'SUCCESS' }, paymentId: 'MOCK_PAYMENT_123', paymentUrl: 'https://mock-paypay.example.com/pay' }) }); } else if (url.includes('/payment-details')) { // 決済完了レスポンス await route.fulfill({ status: 200, contentType: 'application/json', body: JSON.stringify({ resultInfo: { code: 'SUCCESS' }, status: 'COMPLETED', paymentId: 'MOCK_PAYMENT_123' }) }); } }); // PayPay決済フローをテスト const checkout = new CheckoutPage(page); // ... カート追加・配送先入力 await checkout.selectPaymentMethod('paypay'); await page.locator('[data-testid="submit-order"]').click(); // PayPay画面への遷移（モック）を確認 await expect(page.getByText('PayPayで支払う')).toBeVisible(); });

📊 金額計算の整合性テスト

test.describe('金額計算整合性', () => { test('税込金額が正しく計算される', async ({ page }) => { // 商品: 1,000円（税抜）を3個 const checkout = new CheckoutPage(page); const subtotal = await page.locator('[data-testid="subtotal"]').textContent(); const tax = await page.locator('[data-testid="tax"]').textContent(); const total = await checkout.getOrderTotal(); // 小計: 3,000円 expect(subtotal).toContain('3,000'); // 消費税（10%）: 300円 expect(tax).toContain('300'); // 合計: 3,300円 expect(total).toBe(3300); }); test('クーポン適用後の金額が正しい', async ({ page }) => { // 10%OFFクーポン適用 await page.locator('[data-testid="coupon-input"]').fill('SAVE10'); await page.locator('[data-testid="apply-coupon"]').click(); await expect(page.getByText('クーポン適用: -300円')).toBeVisible(); const checkout = new CheckoutPage(page); const total = await checkout.getOrderTotal(); // 3,300円 - 300円 = 3,000円 expect(total).toBe(3000); }); test('送料が正しく加算される', async ({ page }) => { // 北海道への送料 await page.locator('select[data-testid="prefecture"]').selectOption('北海道'); const shipping = await page.locator('[data-testid="shipping-fee"]').textContent(); expect(shipping).toContain('1,200'); // 北海道送料 const checkout = new CheckoutPage(page); const total = await checkout.getOrderTotal(); expect(total).toBe(4500); // 3,300円 + 1,200円 }); test('ポイント利用が正しく反映される', async ({ page }) => { // 500ポイント利用 await page.locator('[data-testid="use-points"]').fill('500'); await page.locator('[data-testid="apply-points"]').click(); await expect(page.getByText('ポイント利用: -500円')).toBeVisible(); const checkout = new CheckoutPage(page); const total = await checkout.getOrderTotal(); expect(total).toBe(2800); // 3,300円 - 500円 }); });
