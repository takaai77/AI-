📊 Core Web Vitals とは

Googleが定義した、ユーザー体験の品質を測る指標です。

LCP
最大コンテンツ描画
< 2.5秒
FID
初回入力遅延
< 100ms
CLS
累積レイアウトシフト
< 0.1
TTFB
最初のバイトまでの時間
< 800ms

📏 パフォーマンス計測の実装

基本的な計測

import { test, expect } from '@playwright/test'; test('ページ読み込みパフォーマンス計測', async ({ page }) => { // Performance APIを使用 await page.goto('https://example-ec.com'); // ナビゲーションタイミングを取得 const performanceData = await page.evaluate(() => { const nav = performance.getEntriesByType('navigation')[0] as PerformanceNavigationTiming; return { // DNS解決時間 dnsLookup: nav.domainLookupEnd - nav.domainLookupStart, // TCP接続時間 tcpConnect: nav.connectEnd - nav.connectStart, // TTFB (Time to First Byte) ttfb: nav.responseStart - nav.requestStart, // ダウンロード時間 download: nav.responseEnd - nav.responseStart, // DOM解析時間 domParsing: nav.domInteractive - nav.responseEnd, // DOMContentLoaded domContentLoaded: nav.domContentLoadedEventEnd - nav.fetchStart, // 完全読み込み loadComplete: nav.loadEventEnd - nav.fetchStart }; }); console.log('Performance Metrics:', performanceData); // しきい値チェック expect(performanceData.ttfb).toBeLessThan(800); expect(performanceData.loadComplete).toBeLessThan(3000); });

Core Web Vitals計測

import { test, expect } from '@playwright/test'; test('Core Web Vitals計測', async ({ page }) => { // web-vitalsライブラリを注入して計測 await page.addInitScript(() => { (window as any).webVitals = { lcp: 0, fid: 0, cls: 0 }; // LCP (Largest Contentful Paint) new PerformanceObserver((list) => { const entries = list.getEntries(); const lastEntry = entries[entries.length - 1]; (window as any).webVitals.lcp = lastEntry.startTime; }).observe({ type: 'largest-contentful-paint', buffered: true }); // CLS (Cumulative Layout Shift) let clsValue = 0; new PerformanceObserver((list) => { for (const entry of list.getEntries()) { if (!(entry as any).hadRecentInput) { clsValue += (entry as any).value; } } (window as any).webVitals.cls = clsValue; }).observe({ type: 'layout-shift', buffered: true }); // FID (First Input Delay) - インタラクション時に計測 new PerformanceObserver((list) => { const firstInput = list.getEntries()[0]; (window as any).webVitals.fid = (firstInput as any).processingStart - firstInput.startTime; }).observe({ type: 'first-input', buffered: true }); }); await page.goto('https://example-ec.com'); // ページが安定するまで待機 await page.waitForLoadState('networkidle'); await page.waitForTimeout(2000); // CLS計測のため // Core Web Vitalsを取得 const vitals = await page.evaluate(() => (window as any).webVitals); console.log('Core Web Vitals:', { LCP: `${vitals.lcp.toFixed(0)}ms`, CLS: vitals.cls.toFixed(3), FID: `${vitals.fid.toFixed(0)}ms` }); // Google推奨しきい値でチェック expect(vitals.lcp).toBeLessThan(2500); // LCP < 2.5秒 expect(vitals.cls).toBeLessThan(0.1); // CLS < 0.1 });

🔍 リソース分析

test('リソース読み込み分析', async ({ page }) => { const resourceData: any[] = []; // リクエストを監視 page.on('response', async (response) => { const timing = response.request().timing(); resourceData.push({ url: response.url(), status: response.status(), size: (response.headers()['content-length'] || '0'), responseTime: timing.responseEnd - timing.requestStart, type: response.request().resourceType() }); }); await page.goto('https://example-ec.com'); await page.waitForLoadState('networkidle'); // リソース分析 const analysis = { totalRequests: resourceData.length, byType: {} as Record<string, number>, slowResources: resourceData .filter(r => r.responseTime > 500) .map(r => ({ url: r.url, time: r.responseTime })), largeResources: resourceData .filter(r => parseInt(r.size) > 100000) .map(r => ({ url: r.url, size: r.size })) }; // タイプ別集計 resourceData.forEach(r => { analysis.byType[r.type] = (analysis.byType[r.type] || 0) + 1; }); console.log('Resource Analysis:', analysis); // パフォーマンス基準チェック expect(analysis.totalRequests).toBeLessThan(100); // リクエスト数 expect(analysis.slowResources.length).toBeLessThan(5); // 遅いリソース });

🚦 Lighthouse連携

// playwright.config.ts でLighthouse連携 import { test as base } from '@playwright/test'; import { playAudit } from 'playwright-lighthouse'; import { chromium } from 'playwright'; const test = base.extend({ lighthousePage: async ({}, use) => { const browser = await chromium.launch({ args: ['--remote-debugging-port=9222'] }); const page = await browser.newPage(); await use(page); await browser.close(); } }); test('Lighthouseスコア計測', async ({ lighthousePage }) => { await lighthousePage.goto('https://example-ec.com'); const result = await playAudit({ page: lighthousePage, port: 9222, thresholds: { performance: 80, accessibility: 90, 'best-practices': 80, seo: 80 } }); console.log('Lighthouse Scores:', { performance: result.lhr.categories.performance.score * 100, accessibility: result.lhr.categories.accessibility.score * 100, bestPractices: result.lhr.categories['best-practices'].score * 100, seo: result.lhr.categories.seo.score * 100 }); });

📈 回帰テスト自動化

// performance-regression.spec.ts import { test, expect } from '@playwright/test'; import * as fs from 'fs'; interface PerformanceBaseline { lcp: number; cls: number; ttfb: number; loadComplete: number; } // ベースライン（前回の計測値） const baseline: PerformanceBaseline = JSON.parse( fs.readFileSync('./performance-baseline.json', 'utf-8') ); // 許容する回帰率（10%） const REGRESSION_THRESHOLD = 1.1; test.describe('パフォーマンス回帰テスト', () => { test('トップページのパフォーマンスが回帰していない', async ({ page }) => { await page.goto('https://example-ec.com'); await page.waitForLoadState('networkidle'); const metrics = await page.evaluate(() => { const nav = performance.getEntriesByType('navigation')[0] as PerformanceNavigationTiming; return { ttfb: nav.responseStart - nav.requestStart, loadComplete: nav.loadEventEnd - nav.fetchStart }; }); // 回帰チェック expect(metrics.ttfb).toBeLessThan(baseline.ttfb * REGRESSION_THRESHOLD); expect(metrics.loadComplete).toBeLessThan(baseline.loadComplete * REGRESSION_THRESHOLD); // 結果をレポート console.log('Performance Comparison:', { ttfb: { baseline: baseline.ttfb, current: metrics.ttfb, change: `${((metrics.ttfb / baseline.ttfb - 1) * 100).toFixed(1)}%` }, loadComplete: { baseline: baseline.loadComplete, current: metrics.loadComplete, change: `${((metrics.loadComplete / baseline.loadComplete - 1) * 100).toFixed(1)}%` } }); }); test('商品一覧ページの読み込み速度', async ({ page }) => { const start = Date.now(); await page.goto('https://example-ec.com/products'); await page.waitForSelector('[data-testid="product-grid"]'); const loadTime = Date.now() - start; // 絶対値でのチェック expect(loadTime).toBeLessThan(3000); // 3秒以内 }); }); // ベースライン更新スクリプト test('ベースライン更新', async ({ page }) => { test.skip(!process.env.UPDATE_BASELINE, 'ベースライン更新モードでない'); await page.goto('https://example-ec.com'); await page.waitForLoadState('networkidle'); const newBaseline = await page.evaluate(() => { const nav = performance.getEntriesByType('navigation')[0] as PerformanceNavigationTiming; return { ttfb: nav.responseStart - nav.requestStart, loadComplete: nav.loadEventEnd - nav.fetchStart, lcp: 0, // 別途計測 cls: 0 }; }); fs.writeFileSync( './performance-baseline.json', JSON.stringify(newBaseline, null, 2) ); });
