❓ Synthetic Monitoringとは

実際のユーザー操作をシミュレートして、本番サイトの稼働状況を継続的に監視する手法です。

🛒 EC監視の重要クリティカルパス

⚠️ 100億ECで絶対に止めてはいけないパス

  • トップページ表示 - サイトの入り口
  • 商品検索 - 購買への導線
  • 商品詳細表示 - 購入判断の場
  • カート追加 - 購入の第一歩
  • 決済完了 - 売上直結

📝 監視スクリプト実装

基本構成

// monitoring/ec-health-check.ts import { chromium, Browser, Page } from 'playwright'; interface HealthCheckResult { checkName: string; status: 'pass' | 'fail'; responseTime: number; error?: string; timestamp: Date; } class ECHealthMonitor { private browser: Browser | null = null; private baseUrl: string; private timeout: number; constructor(baseUrl: string, timeout = 30000) { this.baseUrl = baseUrl; this.timeout = timeout; } async initialize() { this.browser = await chromium.launch({ headless: true, args: ['--no-sandbox', '--disable-dev-shm-usage'] }); } async runAllChecks(): Promise<HealthCheckResult[]> { const results: HealthCheckResult[] = []; results.push(await this.checkTopPage()); results.push(await this.checkSearch()); results.push(await this.checkProductDetail()); results.push(await this.checkAddToCart()); results.push(await this.checkCheckoutFlow()); return results; } // トップページ監視 async checkTopPage(): Promise<HealthCheckResult> { return await this.executeCheck('TopPage', async (page) => { await page.goto(this.baseUrl); await page.waitForSelector('[data-testid="main-content"]'); // 主要要素の存在確認 await page.waitForSelector('[data-testid="search-box"]'); await page.waitForSelector('[data-testid="category-nav"]'); }); } // 検索機能監視 async checkSearch(): Promise<HealthCheckResult> { return await this.executeCheck('Search', async (page) => { await page.goto(this.baseUrl); await page.fill('[data-testid="search-box"]', 'テスト商品'); await page.click('[data-testid="search-submit"]'); // 検索結果が表示されることを確認 await page.waitForSelector('[data-testid="search-results"]'); }); } // 商品詳細監視 async checkProductDetail(): Promise<HealthCheckResult> { return await this.executeCheck('ProductDetail', async (page) => { // 監視用商品ページ await page.goto(`${this.baseUrl}/products/monitoring-test-item`); await page.waitForSelector('[data-testid="product-title"]'); await page.waitForSelector('[data-testid="product-price"]'); await page.waitForSelector('[data-testid="add-to-cart-btn"]'); }); } // カート追加監視 async checkAddToCart(): Promise<HealthCheckResult> { return await this.executeCheck('AddToCart', async (page) => { await page.goto(`${this.baseUrl}/products/monitoring-test-item`); await page.click('[data-testid="add-to-cart-btn"]'); // カートに追加されたことを確認 await page.waitForSelector('[data-testid="cart-notification"]'); }); } // 購入フロー監視（決済直前まで） async checkCheckoutFlow(): Promise<HealthCheckResult> { return await this.executeCheck('CheckoutFlow', async (page) => { // テスト用カートページへ直接遷移 await page.goto(`${this.baseUrl}/cart?monitoring=true`); await page.click('[data-testid="proceed-checkout"]'); // チェックアウト画面が表示されることを確認 await page.waitForSelector('[data-testid="checkout-form"]'); }); } // 汎用チェック実行 private async executeCheck( name: string, checkFn: (page: Page) => Promise<void> ): Promise<HealthCheckResult> { const start = Date.now(); const context = await this.browser!.newContext(); const page = await context.newPage(); page.setDefaultTimeout(this.timeout); try { await checkFn(page); return { checkName: name, status: 'pass', responseTime: Date.now() - start, timestamp: new Date() }; } catch (error) { return { checkName: name, status: 'fail', responseTime: Date.now() - start, error: error.message, timestamp: new Date() }; } finally { await context.close(); } } async close() { await this.browser?.close(); } }

🚨 Slack通知の実装

// monitoring/slack-notifier.ts interface SlackMessage { channel: string; text: string; attachments?: Array<{ color: string; title: string; fields: Array<{ title: string; value: string; short: boolean }>; }>; } class SlackNotifier { private webhookUrl: string; constructor(webhookUrl: string) { this.webhookUrl = webhookUrl; } async sendAlert(results: HealthCheckResult[]) { const failures = results.filter(r => r.status === 'fail'); if (failures.length === 0) return; const message: SlackMessage = { channel: '#ec-alerts', text: `🚨 EC監視アラート: ${failures.length}件の障害検知`, attachments: failures.map(f => ({ color: 'danger', title: `❌ ${f.checkName} - 失敗`, fields: [ { title: 'エラー内容', value: f.error || 'Unknown', short: false }, { title: '応答時間', value: `${f.responseTime}ms`, short: true }, { title: '検知時刻', value: f.timestamp.toISOString(), short: true } ] })) }; await fetch(this.webhookUrl, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(message) }); } async sendRecovery(checkName: string) { const message: SlackMessage = { channel: '#ec-alerts', text: `✅ 復旧: ${checkName} が正常に戻りました` }; await fetch(this.webhookUrl, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(message) }); } }

// monitoring/metrics-reporter.ts import { HealthCheckResult } from './types'; class MetricsReporter { private datadogApiKey: string; constructor(apiKey: string) { this.datadogApiKey = apiKey; } async sendMetrics(results: HealthCheckResult[]) { const timestamp = Math.floor(Date.now() / 1000); const metrics = results.flatMap(r => [ // 可用性（0=失敗, 1=成功） { metric: 'ec.synthetic.availability', points: [[timestamp, r.status === 'pass' ? 1 : 0]], tags: [`check:${r.checkName}`, 'env:production'] }, // レスポンスタイム { metric: 'ec.synthetic.response_time', points: [[timestamp, r.responseTime]], tags: [`check:${r.checkName}`, 'env:production'] } ]); await fetch('https://api.datadoghq.com/api/v1/series', { method: 'POST', headers: { 'Content-Type': 'application/json', 'DD-API-KEY': this.datadogApiKey }, body: JSON.stringify({ series: metrics }) }); } // SLO計算（過去30日の可用性） async calculateSLO(): Promise<{ availability: number; target: number }> { // Datadogからメトリクスを取得して計算 const thirtyDaysAgo = Math.floor(Date.now() / 1000) - (30 * 24 * 60 * 60); const now = Math.floor(Date.now() / 1000); // クエリ例: 過去30日の成功率 const query = `avg:ec.synthetic.availability{env:production}.rollup(avg, 86400)`; // 実際のSLO計算ロジック... return { availability: 99.95, // 実績 target: 99.9 // 目標 }; } }
